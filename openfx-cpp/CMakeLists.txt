# openfx-cpp: header-only C++ bindings over the OFX C API, common to both
# sides (openfx/) with a plugin half (openfx/plugin/) and a host half
# (openfx/host/). Consumers link OpenFX::openfx-cpp and include <openfx/...>.
#
# This file installs the headers and compile-checks them; tests/ has their
# unit tests.

include(GNUInstallDirs)

set(OFX_CPP_HEADER_DIR "${CMAKE_CURRENT_SOURCE_DIR}/include")

add_library(openfx-cpp INTERFACE)
add_library(OpenFX::openfx-cpp ALIAS openfx-cpp)

target_compile_features(openfx-cpp INTERFACE cxx_std_17)
target_include_directories(
  openfx-cpp
  INTERFACE $<BUILD_INTERFACE:${OFX_CPP_HEADER_DIR}>
            $<BUILD_INTERFACE:${OFX_HEADER_DIR}>
            $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)

# openfx/ofxSpan.h uses <span> from C++20 on, and tcbrindle/span below that.
# The conanfile requires tcb-span unless build_openfx_cpp is off.
find_package(tcb-span QUIET)
if(TARGET tcb-span::tcb-span)
  target_link_libraries(openfx-cpp INTERFACE tcb-span::tcb-span)
endif()

# The headers install beside the C API headers, so <openfx/plugin/ofxEffect.h>
# and <ofxImageEffect.h> both resolve from one include directory.
install(
  DIRECTORY "${OFX_CPP_HEADER_DIR}/openfx"
  DESTINATION "${CMAKE_INSTALL_INCLUDEDIR}"
  FILES_MATCHING
  PATTERN "*.h")

if(OFX_BUILD_OPENFX_CPP_TESTS)
  add_subdirectory(tests)
endif()

if(NOT OFX_BUILD_OPENFX_CPP_CHECK)
  return()
endif()

# ---------------------------------------------------------------------------
# openfx-cpp-check: compile every header, on each side, at each standard.
#
# Two generated translation units -- the common headers plus the plugin ones,
# and the common headers plus the host ones -- and the minimal example plugin.
# Each header is included on a line of its own, so one that does not stand
# alone (a missing include, a stale filename) fails the build rather than
# waiting for a consumer to find it.
# ---------------------------------------------------------------------------

file(GLOB OFX_CPP_COMMON_HEADERS CONFIGURE_DEPENDS
     "${OFX_CPP_HEADER_DIR}/openfx/*.h")
file(GLOB OFX_CPP_PLUGIN_HEADERS CONFIGURE_DEPENDS
     "${OFX_CPP_HEADER_DIR}/openfx/plugin/*.h")
file(GLOB OFX_CPP_HOST_HEADERS CONFIGURE_DEPENDS
     "${OFX_CPP_HEADER_DIR}/openfx/host/*.h")

set(OFX_CPP_CHECK_DIR "${CMAKE_CURRENT_BINARY_DIR}/check")

# Write OUT_FILE including each of the remaining arguments as <openfx/...>.
# It goes through a staging file so an unchanged unit is not rewritten, which
# would rebuild the check on every configure.
function(ofx_cpp_write_check_tu OUT_FILE)
  set(text "// Generated by openfx-cpp/CMakeLists.txt. Do not edit.\n")
  foreach(header IN LISTS ARGN)
    file(RELATIVE_PATH rel "${OFX_CPP_HEADER_DIR}" "${header}")
    string(APPEND text "#include <${rel}>\n")
  endforeach()
  file(WRITE "${OUT_FILE}.staged" "${text}")
  configure_file("${OUT_FILE}.staged" "${OUT_FILE}" COPYONLY)
endfunction()

ofx_cpp_write_check_tu(
  "${OFX_CPP_CHECK_DIR}/check-plugin.cpp" ${OFX_CPP_COMMON_HEADERS}
  ${OFX_CPP_PLUGIN_HEADERS})
ofx_cpp_write_check_tu("${OFX_CPP_CHECK_DIR}/check-host.cpp"
                       ${OFX_CPP_COMMON_HEADERS} ${OFX_CPP_HOST_HEADERS})

set(OFX_CPP_CHECK_STANDARDS 17)
if(cxx_std_20 IN_LIST CMAKE_CXX_COMPILE_FEATURES)
  list(APPEND OFX_CPP_CHECK_STANDARDS 20)
endif()
if(NOT TARGET tcb-span::tcb-span)
  list(REMOVE_ITEM OFX_CPP_CHECK_STANDARDS 17)
  message(
    STATUS
      "openfx-cpp: tcb-span not found, skipping the C++17 compile check "
      "(openfx/ofxSpan.h needs it below C++20)")
endif()

add_custom_target(openfx-cpp-check)
set_property(TARGET openfx-cpp-check PROPERTY FOLDER openfx-cpp)

# An object library: compiling is the whole check, so nothing is linked.
function(ofx_cpp_add_check NAME SOURCE STANDARD)
  set(target "openfx-cpp-check-${NAME}-cxx${STANDARD}")
  add_library(${target} OBJECT "${SOURCE}")
  target_link_libraries(${target} PRIVATE OpenFX::openfx-cpp)
  set_target_properties(
    ${target} PROPERTIES CXX_STANDARD ${STANDARD} CXX_STANDARD_REQUIRED ON
                         CXX_EXTENSIONS OFF FOLDER openfx-cpp)
  add_dependencies(openfx-cpp-check ${target})
endfunction()

foreach(standard IN LISTS OFX_CPP_CHECK_STANDARDS)
  ofx_cpp_add_check(plugin "${OFX_CPP_CHECK_DIR}/check-plugin.cpp" ${standard})
  ofx_cpp_add_check(host "${OFX_CPP_CHECK_DIR}/check-host.cpp" ${standard})
  ofx_cpp_add_check(
    minimal-plugin "${CMAKE_CURRENT_SOURCE_DIR}/examples/minimal-plugin/minimal.cpp"
    ${standard})
endforeach()
